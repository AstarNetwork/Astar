19//! # Contracts MBM
20//!
21//! Multi block migrations decommissioning `pallet-contracts` (ink!/Wasm smart contracts) from the
22//! Astar, Shiden & Shibuya runtimes. Staged over two runtime upgrades:
23//!
24//! 1. [`ReleaseContractsDeposits`] - settles every balance the pallet still holds. Must complete
25//!    before the pallet index is dropped, otherwise those holds become undecodable and the funds
26//!    are frozen for good.
27//! 2. *(next runtime)* purging the contract child tries and the pallet prefix.
28

48#[frame_support::pallet]
49pub mod pallet {
50    use frame_support::traits::Get;
51
52    /// Carries no storage and no calls. It exists only so that the migrations in this crate can
53    /// be benchmarked through the standard `benchmark pallet` tooling, and is therefore added to
54    /// the runtimes under `runtime-benchmarks` only.
55    #[pallet::pallet]
56    #[pallet::without_storage_info]
57    pub struct Pallet<T>(_);
58
59    #[pallet::config]
60    pub trait Config: frame_system::Config + pallet_balances::Config {
61        /// Hold reason the `release_deposit` benchmark places on its sample account.
62        type BenchmarkHoldReason: Get<<Self as pallet_balances::Config>::RuntimeHoldReason>;
63    }
64}
