19//! Settles the balances `pallet-contracts` still owns, so that its pallet index can be retired.
20
21use frame_support::{
22    migrations::{SteppedMigration, SteppedMigrationError},
23    traits::{
24        fungible::MutateHold,
25        tokens::{Fortitude, Precision, Restriction},
26        Get,
27    },
28    weights::WeightMeter,
29};
30use sp_io::hashing::blake2_256;
31use sp_runtime::traits::Zero;
32use sp_std::marker::PhantomData;

34use crate::{WeightInfo, LOG_TARGET};
35
36type HoldReasonOf<T> = <T as pallet_balances::Config>::RuntimeHoldReason;
37type BalanceOf<T> = <T as pallet_balances::Config>::Balance;
38
39/// Moves every balance hold placed by `pallet-contracts` to an escrow account.
40///
41/// # Why this has to happen *before* the pallet is removed
42///
43/// Storage and code deposits are live `Balances::Holds` entries whose reason is
44/// `RuntimeHoldReason::Contracts(..)`, i.e. tagged with the pallet's index. Dropping the pallet
45/// from `construct_runtime` deletes that variant and every such entry becomes undecodable.
46/// `Holds` is a `ValueQuery` map, so the decode failure silently yields an empty vector: the hold
47/// record disappears while `AccountData::reserved` stays behind. Nothing can decrement that
48/// `reserved` again, so the funds are frozen for good - and silently, because
49/// `pallet_balances::do_try_state` only bounds the length of each hold vector, it never checks
50/// that the holds sum to `reserved`. Hence the `post_upgrade` check below.
51///
52/// # Why everything goes to escrow
53///
54/// Storage deposits sit on keyless, derived contract accounts, so releasing them in place would
55/// strand the funds; the depositor is not recorded on chain. Code upload deposits do sit on real
56/// accounts, but sweeping both to a single escrow account keeps this migration to one code path
57/// and leaves the refunding to a plain, auditable batch of transfers afterwards.
58///
59/// # What this deliberately does not touch
60///
61/// The *free* balance of contract accounts is contract owned value, not a deposit - for a vault
62/// or DEX contract it is user funds whose rightful owner is encoded in the contract's own child
63/// trie, not "whoever deployed it". Sweeping it is a policy decision, not a technical necessity:
64/// unlike the holds it breaks no invariant if left in place, and `Balances::force_transfer` from
65/// Root can still move it at any point in the future. It is deliberately left for the migration
66/// that purges `ContractInfoOf`, which is the last on-chain record of which accounts are
67/// contracts.
68pub struct ReleaseContractsDeposits<T, CodeDeposit, StorageDeposit, Escrow, W>(
69    PhantomData<(T, CodeDeposit, StorageDeposit, Escrow, W)>,
70);
71
72impl<T, CodeDeposit, StorageDeposit, Escrow, W> SteppedMigration
73    for ReleaseContractsDeposits<T, CodeDeposit, StorageDeposit, Escrow, W>
74where
75    T: pallet_balances::Config,
76    HoldReasonOf<T>: PartialEq,
77    CodeDeposit: Get<HoldReasonOf<T>>,
78    StorageDeposit: Get<HoldReasonOf<T>>,
79    Escrow: Get<T::AccountId>,
80    W: WeightInfo,
81{
82    /// Last account whose holds were settled.
83    type Cursor = T::AccountId;
84    type Identifier = [u8; 32];
85
86    fn id() -> Self::Identifier {
87        blake2_256(b"contracts-mbm::ReleaseContractsDeposits")
88    }
89
90    fn step(
91        cursor: Option<Self::Cursor>,
92        meter: &mut WeightMeter,
93    ) -> Result<Option<Self::Cursor>, SteppedMigrationError> {
94        let required = W::release_deposit();
95        // Guarantees the loop below settles at least one account, so a step never returns the
96        // cursor it was handed and stalls the migration forever.
97        if meter.remaining().any_lt(required) {
98            return Err(SteppedMigrationError::InsufficientWeight { required });
99        }
100
101        let reasons = [CodeDeposit::get(), StorageDeposit::get()];
102        let escrow = Escrow::get();
103        // Removing the entry of the account currently being visited is safe: the iterator has
104        // already recorded its key to seek from.
105        let mut iter = match &cursor {
106            Some(last) => pallet_balances::Holds::<T>::iter_from(
107                pallet_balances::Holds::<T>::hashed_key_for(last),
108            ),
109            None => pallet_balances::Holds::<T>::iter(),
110        };
111        let mut last = cursor;
112        let mut settled = 0u32;
113
114        loop {
115            if !meter.can_consume(required) {
116                return Ok(last);
117            }
118
119            let Some((account, holds)) = iter.next() else {
120                log::info!(
121                    target: LOG_TARGET,
122                    "ReleaseContractsDeposits: finished, {settled} accounts settled in this step 💸",
123                );
124                return Ok(None);
125            };
126
127            // Charged whether or not the account turns out to be relevant: the read happened
128            // either way, and this keeps the loop guaranteed to make progress.
129            meter.consume(required);
130            settled = settled.saturating_add(1);
131            last = Some(account.clone());
132
133            for hold in holds.iter() {
134                if hold.amount.is_zero() || !reasons.contains(&hold.id) {
135                    continue;
136                }
137                Self::sweep_to_escrow(&account, &escrow, hold.amount, &hold.id);
138            }
139        }
140    }
141
142    #[cfg(feature = "try-runtime")]
143    fn post_upgrade(_state: sp_std::vec::Vec<u8>) -> Result<(), sp_runtime::TryRuntimeError> {
144        let reasons = [CodeDeposit::get(), StorageDeposit::get()];
145
146        for (_, holds) in pallet_balances::Holds::<T>::iter() {
147            if holds.iter().any(|hold| reasons.contains(&hold.id)) {
148                return Err(
149                    "A pallet-contracts hold survived; removing the pallet index would \
150                            freeze these funds permanently 🚨"
151                        .into(),
152                );
153            }
154        }
155
156        Ok(())
157    }
158}
159
160impl<T, CodeDeposit, StorageDeposit, Escrow, W>
161    ReleaseContractsDeposits<T, CodeDeposit, StorageDeposit, Escrow, W>
162where
163    T: pallet_balances::Config,
164{
165    fn sweep_to_escrow(
166        account: &T::AccountId,
167        escrow: &T::AccountId,
168        amount: BalanceOf<T>,
169        reason: &HoldReasonOf<T>,
170    ) {
171        match pallet_balances::Pallet::<T>::transfer_on_hold(
172            reason,
173            account,
174            escrow,
175            amount,
176            Precision::BestEffort,
177            Restriction::Free,
178            // The source is typically a keyless contract account; the existential deposit must
179            // not stand in the way of recovering the funds.
180            Fortitude::Force,
181        ) {
182            Ok(moved) => log::info!(
183                target: LOG_TARGET,
184                "ReleaseContractsDeposits: swept {moved:?} from {account:?} to the escrow",
185            ),
186            // Failing the whole migration would leave the remaining holds in place, which is
187            // exactly the outcome this is meant to avoid. `post_upgrade` catches any leftover.
188            Err(err) => log::error!(
189                target: LOG_TARGET,
190                "ReleaseContractsDeposits: failed to sweep {amount:?} from {account:?}: {err:?}",
191            ),
192        }
193    }
